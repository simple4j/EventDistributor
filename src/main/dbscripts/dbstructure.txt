create table EVENT
(
EVENT_ID number(10),
BUSINESS_RECORD_ID varchar2(128) not null,
BUSINESS_REPORT_TYPE varchar2(64) not null,
BUSINESS_RECORD_SUB_TYPE varchar2(64),
BUSINESS_RECORD_VERSION varchar2(64),
SOURCE varchar2(32) not null,
STATUS varchar2(16) not null,
DUPLICATE_EVENT_ID number(10),
DUPLICATE_CHECK_END_TIME timestamp with local time zone,
REPOST_PARENT_EVENT_ID number(10),
PROCESSING_HOST varchar2(64),
STATUS_EXPIRY_TIME timestamp with local time zone,
CREATE_TIME timestamp with local time zone not null,
UPDATE_TIME timestamp with local time zone not null,
CREATE_BY varchar2(64),
UPDATE_BY varchar2(64),
CONSTRAINT EVENT_PK primary key (EVENT_ID),
CONSTRAINT EVENT_FK1
FOREIGN KEY(DUPLICATE_EVENT_ID) REFERENCES EVENT(EVENT_ID) 
    ON DELETE CASCADE,
CONSTRAINT EVENT_FK2
FOREIGN KEY(REPOST_PARENT_EVENT_ID) REFERENCES EVENT(EVENT_ID) 
    ON DELETE CASCADE
);

CREATE INDEX EVENT_IDX1
ON EVENT(DUPLICATE_EVENT_ID);

CREATE INDEX EVENT_IDX2
ON EVENT(REPOST_PARENT_EVENT_ID);

CREATE INDEX EVENT_IDX3
ON EVENT(DUPLICATE_CHECK_END_TIME);

CREATE INDEX EVENT_IDX4
ON EVENT(BUSINESS_RECORD_ID);

CREATE INDEX EVENT_IDX5
ON EVENT(STATUS_EXPIRY_TIME);

create table EVENT_TARGET
(
EVENT_ID number(10),
TARGET_ID varchar2(32),
CONSTRAINT EVENT_TARGET_PK primary key (EVENT_ID,TARGET_ID),
CONSTRAINT EVENT_TARGET_FK
FOREIGN KEY(EVENT_ID) REFERENCES EVENT(EVENT_ID) 
    ON DELETE CASCADE
)

CREATE INDEX EVENT_TARGET_IDX1
ON EVENT_TARGET(EVENT_ID);
